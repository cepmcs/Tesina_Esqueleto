% =========================================================================
% GUÍA Y PALETA DE COLORES PARA FIGURAS DE LA TESINA (OKABE-ITO 4 COLORES)
% =========================================================================
% Toda figura, diagrama TikZ o gráfico de esta tesina debe usar esta paleta.
% La muestra visual compilada se encuentra en Figures/paleta_colores.pdf
%
% ROLES SEMÁNTICOS:
% 1. figBlue      (#0072B2) / figBlueTint      (#EBF3F9) -> Primario, método propuesto, flujo base / Serie 1
% 2. figOrange    (#E69F00) / figOrangeTint    (#FEF6E9) -> Acento, contraste, elementos destacados / Serie 2
% 3. figGreen     (#009E73) / figGreenTint     (#EAF7F3) -> Validación positiva, éxito, métricas / Serie 3
% 4. figVermilion (#D55E00) / figVermilionTint (#FDF0EC) -> Pérdida (loss), restricciones, alertas / Serie 4
%
% REGLAS DE DISEÑO PARA NODOS TIKZ:
% - Bordes/líneas: color sólido (ej. draw=figBlue, line width=1pt a 1.2pt)
% - Fondo de cajas: color suave (ej. fill=figBlueTint)
% - Texto de nodos: siempre text=darkText (#1A202C) para asegurar contraste
% =========================================================================

% 3. Verde azulado (Bluish Green) - Validación positiva / Éxito / Serie 3
\definecolor{figGreen}{HTML}{009E73}
\definecolor{figGreenTint}{HTML}{EAF7F3}

\definecolor{figVermilionTint}{HTML}{FDF0EC}
